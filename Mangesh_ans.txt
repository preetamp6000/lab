-- Create the department table
CREATE TABLE dept (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50)
);

-- Create the employee table
CREATE TABLE emp (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(50),
    dept_id INT,
    salary INT,
    manager_id INT,
    FOREIGN KEY (dept_id) REFERENCES dept(dept_id)
);

SELECT 
    emp_name, 
    COALESCE((SELECT dept_name FROM dept WHERE dept.dept_id = emp.dept_id), 'No Department') AS dept_name
FROM emp;

2.
SELECT 
    dept_name, 
    (SELECT COUNT(emp_id) FROM emp WHERE emp.dept_id = dept.dept_id) AS total_employees
FROM dept;

3.SELECT 
    d.dept_name, 
    e.emp_name
FROM dept d
LEFT JOIN emp e ON d.dept_id = e.dept_id;

4.
SELECT 
    e1.emp_name AS Employee_1, 
    e2.emp_name AS Employee_2 
FROM emp e1
JOIN emp e2 ON e1.dept_id = e2.dept_id 
           AND e1.manager_id = e2.manager_id 
           AND e1.emp_id < e2.emp_id;

5.
SELECT 
    dept_name, 
    (SELECT SUM(salary) FROM emp WHERE emp.dept_id = dept.dept_id) AS total_salary
FROM dept;

6.
SELECT 
    emp_name, 
    salary
FROM emp e1
WHERE salary > (
    SELECT AVG(salary) 
    FROM emp e2 
    WHERE e1.dept_id = e2.dept_id
);
